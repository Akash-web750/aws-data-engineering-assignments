# Runtime packages for the producer and the consumer.

# Kafka client (producer, consumer, admin). Ships pre-built Windows wheels.

# Loads the optional .env file read by config/settings.py.
python-dotenv>=1.0

# PostgreSQL driver, used only by the source_db package (PostgreSQL source

# PostgreSQL client installation is needed.
